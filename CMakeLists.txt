cmake_minimum_required(VERSION 3.22)
project(AnimalAtlas VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ATLAS_BUILD_TESTS "Build portable core tests" ON)
option(ATLAS_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)
option(ATLAS_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" ON)
option(ATLAS_BUILD_AUDIO_HOST "Build the optional JUCE native playback host" OFF)
set(ATLAS_JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE source checkout; no automatic download")

add_library(atlas_core native/src/core.cpp)
target_include_directories(atlas_core PUBLIC native/include)
target_compile_features(atlas_core PUBLIC cxx_std_20)
set_target_properties(atlas_core PROPERTIES CXX_EXTENSIONS OFF)

add_library(atlas_audio native/src/audio.cpp native/src/wav.cpp)
target_link_libraries(atlas_audio PUBLIC atlas_core)

find_package(Threads REQUIRED)
add_library(atlas_playback native/src/playback.cpp)
target_link_libraries(atlas_playback PUBLIC atlas_audio PRIVATE Threads::Threads)

find_package(SQLite3 REQUIRED)
add_library(atlas_storage native/src/storage.cpp native/src/sha256.cpp native/src/event_validation.cpp native/src/ui_host.cpp)
target_link_libraries(atlas_storage PUBLIC atlas_core atlas_playback)
if(TARGET SQLite3::SQLite3)
    target_link_libraries(atlas_storage PRIVATE SQLite3::SQLite3)
else()
    target_link_libraries(atlas_storage PRIVATE SQLite::SQLite3)
endif()

# Identify the compiled production source, including uncommitted development edits.
file(GLOB_RECURSE atlas_identity_sources CONFIGURE_DEPENDS
    native/src/*.cpp native/src/*.hpp native/include/*.hpp native/cmake/*.in)
list(APPEND atlas_identity_sources "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT atlas_identity_sources)
set(atlas_identity_manifest "")
foreach(source IN LISTS atlas_identity_sources)
    file(SHA256 "${source}" digest)
    file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${source}")
    string(APPEND atlas_identity_manifest "${relative}:${digest}\n")
endforeach()
string(SHA256 ATLAS_NATIVE_SOURCE_SHA256 "${atlas_identity_manifest}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${atlas_identity_sources})
configure_file(native/cmake/build_identity.hpp.in generated/atlas_build_identity.hpp @ONLY)

add_executable(atlas-native native/src/main.cpp)
target_include_directories(atlas-native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(atlas-native PRIVATE atlas_storage)

if(ATLAS_BUILD_AUDIO_HOST)
    if(NOT EXISTS "${ATLAS_JUCE_SOURCE_DIR}/modules/juce_audio_devices/juce_audio_devices.h")
        message(FATAL_ERROR "ATLAS_JUCE_SOURCE_DIR must name an existing JUCE checkout")
    endif()
    enable_language(C)
    set(JUCE_MODULES_ONLY ON CACHE BOOL "Use JUCE modules without packaging tools" FORCE)
    add_subdirectory("${ATLAS_JUCE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
    add_executable(atlas-audio-host native/src/audio_host_main.cpp native/src/juce_output.cpp)
    target_link_libraries(atlas-audio-host PRIVATE atlas_storage atlas_playback juce::juce_audio_devices juce::juce_events)
    target_include_directories(atlas-audio-host PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(atlas-audio-host PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1
        JUCE_STANDALONE_APPLICATION=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
endif()

    add_executable(atlas-core-tests native/tests/core_tests.cpp)
    target_link_libraries(atlas-core-tests PRIVATE atlas_core)
    add_test(NAME atlas-core COMMAND atlas-core-tests)
    add_executable(atlas-sha256-tests native/tests/sha256_tests.cpp)
    target_include_directories(atlas-sha256-tests PRIVATE native/src)
    target_link_libraries(atlas-sha256-tests PRIVATE atlas_storage)

    add_executable(atlas-audio-tests native/tests/audio_tests.cpp)
    target_link_libraries(atlas-audio-tests PRIVATE atlas_audio)
    add_test(NAME atlas-audio COMMAND atlas-audio-tests)
    add_executable(atlas-audio-bench native/tests/audio_bench.cpp)
    target_link_libraries(atlas-audio-bench PRIVATE atlas_audio)

set(atlas_targets atlas_core atlas_audio atlas_playback atlas_storage atlas-native)
if(ATLAS_BUILD_AUDIO_HOST)
    list(APPEND atlas_targets atlas-audio-host)
endif()
if(ATLAS_BUILD_TESTS)
    list(APPEND atlas_targets atlas-core-tests atlas-sha256-tests atlas-audio-tests atlas-audio-bench atlas-playback-tests atlas-playback-fixture-host)
endif()
foreach(target IN LISTS atlas_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
        if(ATLAS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
        if(ATLAS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
        if(ATLAS_SANITIZERS)
            target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
